#!/bin/bash
file="addressbook.txt"

while true
do
    echo "----------------------------"
    echo " ADDRESS BOOK "
    echo "----------------------------"
    echo "1. Create Address Book"
    echo "2. View Address Book"
    echo "3. Insert Record"
    echo "4. Delete Record"
    echo "5. Modify Record"
    echo "6. Exit"
    echo "----------------------------"

    read -p "Enter your choice: " choice
    case $choice in
    1)
        > $file
        echo "Address book created successfully."
        ;;
    2)
        if [ -f “$file” ]
        then
            echo "------ Address Book ------"
            cat “$file”
        else
            echo "Address book does not exist."
        fi
        ;;
    3)
        read -p "Enter Name: " name
        read -p "Enter Phone: " phone
        read -p "Enter Email: " email
        read -p "Enter City: " city
        echo "$name|$phone|$email|$city" >> $file
        echo "Record inserted successfully."
        ;;
    4)
        read -p "Enter name to delete: " name
        grep -v "^$name|" $file > temp.txt
        mv temp.txt $file
        echo "Record deleted successfully."
        ;;
    5)
        read -p "Enter name to modify: " name
        grep -v "^$name|" $file > temp.txt
        read -p "Enter New Name: " newname
        read -p "Enter New Phone: " newphone
        read -p "Enter New Email: " newemail
        read -p "Enter New City: " newcity
        echo "$newname|$newphone|$newemail|$newcity" >> temp.txt
        mv temp.txt $file
        echo "Record modified successfully."
        ;;
    6)
        echo "Thank You!"
        exit
        ;;
    *)
        echo "Invalid Choice"
        ;;
    esac
done
