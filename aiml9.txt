import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import importlib

data=pd.read_csv(r"C:\Users\Administrator\Downloads\Mall_Customers Ass9.csv")

data

data.info()

data.shape

data.duplicated().sum()

data.isnull().sum()

X=data[["Annual Income (k$)", "Spending Score (1-100)"]]

X

from sklearn.preprocessing import StandardScaler

scaler=StandardScaler()

x_scaled=scaler.fit_transform(X)

plt = importlib.reload(importlib.import_module("matplotlib.pyplot"))

from sklearn.cluster import KMeans

WCSS=[]
for i in range(1,11):
    kmeans=KMeans(n_clusters=i, random_state=42,n_init=10)
    kmeans.fit(x_scaled)
    WCSS.append(kmeans.inertia_)

plt.plot(range(1,11),WCSS,marker="o")
plt.title("Elbow Method")
plt.xlabel("No of Clusters")
plt.ylabel("WCSS")
plt.show()

kmeans=KMeans(n_clusters=5, random_state=42,n_init=10)
data["KMeans Cluster"]=kmeans.fit_predict(x_scaled)

data

plt.scatter(X["Annual Income (k$)"],X["Spending Score (1-100)"],c=data["KMeans Cluster"],cmap="viridis")
plt.xlabel("Annual Income (k$)")
plt.ylabel("Spending Score (1-100)")
plt.title("KMeans Clusters")
plt.show()

from scipy.cluster.hierarchy import dendrogram,linkage,fcluster 

linked=linkage(x_scaled,method="ward")

plt.figure(figsize=(12,6))
dendrogram(linked)
plt.title("Hierarchical Clustering Dendrogram")
plt.xlabel("Customers")
plt.ylabel("Distance")
plt.show()

data["Hierarchical cluster"]=fcluster(linked,t=5,criterion="maxclust")
data[["Annual Income (k$)","Spending Score (1-100)","Hierarchical cluster"]]

plt.scatter(X["Annual Income (k$)"],X["Spending Score (1-100)"],c=data["Hierarchical cluster"],cmap="viridis")
plt.xlabel("Annual Income (k$)")
plt.ylabel("Spending Score (1-100)")
plt.title("Hierarchical cluster")
plt.show()

analysis=data.groupby("KMeans Cluster").agg(
    Customers=("KMeans Cluster","size"),
    Average_Income=("Annual Income (k$)","mean"),
    Average_Spending=("Spending Score (1-100)","mean"))
print(analysis.round(2))
